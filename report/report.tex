\chapter{TỔNG QUAN VỀ DỰ ÁN}

\section{Giới thiệu chung và Bài toán thực tiễn}
Trong kỷ nguyên chuyển đổi số giáo dục và tự động hóa quy trình nghiệp vụ doanh nghiệp, tài liệu định dạng PDF đóng vai trò là phương tiện lưu chuyển văn bản, đề thi, bảng điểm, giáo trình và tài liệu tài chính phổ biến nhất. Tuy nhiên, thách thức lớn nhất mà các tổ chức đang phải đối mặt chính là \textbf{nguy cơ rò rỉ dữ liệu mật, phát tán trái phép tài liệu có bản quyền và gian lận đề thi} thông qua các kênh trao đổi trực tuyến.

Các phương pháp bảo vệ truyền thống như đặt mật khẩu mở file (Password Protection) hoặc mã hóa tĩnh thường mang lại trải nghiệm người dùng kém, đồng thời hoàn toàn bất lực trước hành vi chụp ảnh màn hình hoặc phát tán tài liệu sau khi đã mở khóa. Để giải quyết triệt để vấn đề này, giải pháp tối ưu là \textbf{cá nhân hóa động (Dynamic Watermarking)}: tự động chèn thông tin định danh của chính người tải (Họ tên, Mã sinh viên, Thời gian tải, Mã giao dịch và Mã băm kiểm định SHA-256) chéo lên toàn bộ các trang của tệp PDF ngay tại thời điểm yêu cầu được gửi tới máy chủ.

Hơn thế nữa, mã băm \textbf{SHA-256} được tính toán độc lập và lưu vào cơ sở dữ liệu kiểm toán (Audit Logs) đảm bảo tính \textbf{chống chối bỏ (Non-Repudiation)} và \textbf{toàn vẹn dữ liệu (Integrity)}: bất kỳ sự can thiệp, chỉnh sửa hay tẩy xóa chữ số nào trên tài liệu đều bị phát hiện ngay lập tức khi tra cứu trên hệ thống.

\section{Đối tượng sử dụng và Hệ sinh thái tích hợp}
Hệ thống được thiết kế theo mô hình \textbf{Nền tảng Ứng dụng Web trực tuyến đa năng (Full-stack Web Platform)}, tích hợp giao diện người dùng Web Studio hiện đại đồng thời cung cấp giao tiếp chuẩn hóa qua \textbf{RESTful JSON API} (\texttt{POST /api/v1/watermark}). Hệ thống hướng tới 05 nhóm đối tượng thực tế:
\begin{itemize}
    \item \textbf{Khối Khảo thí \& Đào tạo Đại học (EdTech / LMS / Online Exams):} Tự động đóng dấu mã sinh viên, ca thi, mã đề thi lên đề thi trực tuyến của hàng nghìn sinh viên cùng một lúc ngay trước giờ thi, ngăn chặn tuyệt đối tình trạng rò rỉ đề thi ra ngoài.
    \item \textbf{Khối Tài chính - Ngân hàng - Fintech (Core Banking, Billing System):} Tự động đóng dấu bản sao kê giao dịch ngân hàng, phiếu thu, hóa đơn điện tử đỏ gắn mã khách hàng và con dấu số điện tử bảo mật.
    \item \textbf{Khối Quản trị Doanh nghiệp (ERP, HRM, CRM):} Xuất bảng lương bảo mật (Payslip), quyết định bổ nhiệm, hợp đồng kinh tế nội bộ, đảm bảo thông tin nhân sự và báo giá nhạy cảm không bị chia sẻ ra bên ngoài.
    \item \textbf{Khối Y tế \& Bảo hiểm số (HIS, InsurTech):} Xuất kết quả xét nghiệm bệnh lý, hồ sơ bệnh án, chứng từ bồi thường bảo hiểm bảo mật thông tin cá nhân.
    \item \textbf{Bên thứ ba thẩm định và cơ quan kiểm định (Audit \& Legal):} Sử dụng cổng tra cứu Web Portal công khai để đối chiếu mã SHA-256 của file PDF nghi vấn với nhật ký cấp phát bất biến lưu trong cơ sở dữ liệu.
\end{itemize}

\section{Lý do bắt buộc sử dụng Điện toán đám mây \& Tính đàn hồi}
Bài toán xử lý đóng dấu PDF hàng loạt sở hữu những đặc thù kỹ thuật rất khắt khe:
\begin{enumerate}
    \item \textbf{Tác vụ tiêu tốn tài nguyên tính toán cực lớn (Compute-Intensive Task):} Đóng dấu PDF đòi hỏi giải mã nhị phân, tính toán ma trận tọa độ xoay chữ $45^\circ$, vẽ đồ họa vector Unicode tiếng Việt lên từng trang, nén tối ưu hóa kích thước tệp và băm mật mã học SHA-256. Đóng dấu một tài liệu 47 trang chiếm gần \textbf{100\% công suất của 1 vCPU trong 0.6 -- 1.2 giây}.
    \item \textbf{Đặc thù lưu lượng mang tính đột biến cực cao (Spike Traffic / Flash Crowds):} Lưu lượng dồn dập vào những mốc thời gian cố định (khi mở đề thi cho 5.000 sinh viên hoặc khi bắn thông báo bảng lương). Nếu chỉ dùng 1 máy chủ, hàng đợi sẽ tắc nghẽn lập tức, CPU đạt 100\%, gây lỗi \textit{Gateway Timeout (504)} hoặc sập máy chủ do cạn RAM.
    \item \textbf{Yêu cầu tính đàn hồi (Elasticity) và Tối ưu chi phí (FinOps):} Thay vì duy trì máy chủ cỡ lớn 24/7, \textbf{AWS Auto Scaling} cho phép bình thường chỉ chạy 2 máy chủ nhỏ (\texttt{t3.small}); khi tải tăng vọt, hệ thống \textbf{tự động mở rộng (Scale Out) lên 4 -- 6 máy chủ trong vài phút}, và tự động thu hồi khi hết tải, giúp \textbf{tiết kiệm hơn 80\% chi phí vận hành hạ tầng}.
    \item \textbf{Độ sẵn sàng cao (High Availability - Multi-AZ):} Cụm EC2 chạy trên 2 Vùng sẵn sàng độc lập (\texttt{ap-southeast-1a} và \texttt{ap-southeast-1b}) đứng sau Application Load Balancer đảm bảo hệ thống không có điểm lỗi đơn (No SPOF).
\end{enumerate}

\section{Các dịch vụ AWS cốt lõi được sử dụng trong dự án}
Hệ thống sử dụng đồng bộ 10 dịch vụ đám mây then chốt của Amazon Web Services:
\begin{itemize}
    \item \textbf{Amazon EC2 \& Auto Scaling Group (ASG):} Cụm Worker xử lý trong Private Subnet, tự động mở rộng theo tải CPU.
    \item \textbf{Application Load Balancer (ALB):} Cân bằng tải Layer 7 tại Public Subnet, kiểm tra sức khỏe qua \texttt{/health} và phân phối Round-Robin.
    \item \textbf{Amazon S3 (Simple Storage Service):} Lưu trữ đối tượng PDF gốc (\texttt{input/}) và tệp kết quả (\texttt{output/}), cung cấp Pre-signed URL và Lifecycle Rule hủy file sau 24h.
    \item \textbf{Amazon EFS (Elastic File System):} Hệ thống tệp mạng chia sẻ NFSv4, cache font Unicode và PDF mẫu gốc với độ trễ \textbf{$< 5$ ms}.
    \item \textbf{Amazon EBS (gp3):} Ổ cứng khối 20 GB cho hệ điều hành và vùng đệm xử lý file tại \texttt{/tmp}.
    \item \textbf{Amazon RDS for PostgreSQL 18.3:} Dịch vụ CSDL quan hệ trong Private Subnet lưu trữ nhật ký kiểm toán \texttt{audit\_logs}.
    \item \textbf{Amazon CloudWatch \& Amazon SNS:} Giám sát chỉ số CPU với chu kỳ 60s, kích hoạt Alarm khi CPU $\ge 50\%$ và gửi email cảnh báo qua SNS.
    \item \textbf{Amazon VPC, NAT Gateway, Security Groups \& IAM Roles:} Thiết kế mạng phân tầng Public/Private an toàn, phân quyền tối thiểu Least Privilege.
\end{itemize}

\chapter{GIẢI PHÁP THỰC HIỆN}

\section{Kiến trúc tính năng và 5 Phân hệ Web Studio Trực quan}
Hệ thống được xây dựng dưới dạng Nền tảng Ứng dụng Web trực tuyến toàn diện bằng \textbf{FastAPI} và thư viện đồ họa vector \textbf{PyMuPDF (fitz)}, tích hợp 05 phân hệ nghiệp vụ:
\begin{itemize}
    \item \textbf{1. Kho Lưu Trữ Tài Liệu Gốc (Master Document Library):} Upload file PDF gốc lên S3 và cache sang EFS (\texttt{/mnt/efs/input/}), quản lý danh mục, xem trước trang đầu, phân loại và xóa tài liệu tập trung.
    \item \textbf{2. Phòng Đóng Dấu Đơn Lẻ (Single Live Studio \& Canvas Preview):} Tùy biến dấu chìm xoay $45^\circ$, độ mờ, font tiếng Việt, tiêu đề Header/Footer và con dấu số. Tích hợp \textit{HTML5 Canvas Preview} xem trước trang in thời gian thực.
    \item \textbf{3. Phòng Đóng Dấu Hàng Loạt \& Nén ZIP Đa Luồng (Batch Studio):} Nạp danh sách qua CSV/Excel, xử lý song song đa luồng (\texttt{ThreadPoolExecutor}), xuất bảng kê \texttt{kiem\_dinh\_manifest.csv} (chứa mã SHA-256 từng bản) và nén thành tệp ZIP tải về qua S3 Pre-signed URL.
    \item \textbf{4. Cổng Kiểm Định Tính Toàn Vẹn Số (Public SHA-256 Verifier):} Kéo thả file PDF, tính mã SHA-256 bằng \textit{Web Crypto API} trong $<20$ ms ngay tại trình duyệt (không cần upload file), đối soát tức thì với RDS PostgreSQL.
    \item \textbf{5. Nhật Ký Kiểm Toán Phân Tán (RDS PostgreSQL Audit Log Viewer):} Giám sát toàn bộ phiên đóng dấu trên các Worker node, tra cứu mã SHA-256, định danh host và mốc thời gian GMT+7.
\end{itemize}

\section{Kiến trúc triển khai hạ tầng trên AWS (Tiered Architecture)}
Hạ tầng được triển khai tại vùng Singapore (\texttt{ap-southeast-1}) theo mô hình phân tầng chuẩn Multi-AZ:
\begin{itemize}
    \item \textbf{Public Subnets (10.0.1.0/24, 10.0.2.0/24):} Chứa Application Load Balancer (Port 80) và NAT Gateway.
    \item \textbf{Private Subnets (10.0.11.0/24, 10.0.12.0/24):} Chứa cụm EC2 Worker (Port 8000), đĩa mạng chia sẻ Amazon EFS và CSDL Amazon RDS PostgreSQL 18.3 (Port 5432). Không gán Public IP.
\end{itemize}

\subsection{Luồng xử lý chi tiết theo vòng đời yêu cầu (Request Lifecycle)}
Mọi yêu cầu xử lý đều trải qua chu trình khép kín 08 bước:
\begin{enumerate}
    \item \textbf{Client $\to$ ALB:} Client gửi \texttt{POST /api/v1/watermark} kèm JSON payload.
    \item \textbf{ALB $\to$ EC2 Worker:} ALB chuyển tiếp HTTP request đến một worker khỏe mạnh qua thuật toán Round-Robin.
    \item \textbf{EC2 $\leftrightarrow$ EFS Storage:} Worker đọc nhanh font Unicode và PDF mẫu gốc từ \texttt{/mnt/efs/input/} ($< 5$ ms).
    \item \textbf{EC2 Engine:} PyMuPDF render watermark xoay $45^\circ$, con dấu đỏ và tính toán mã băm SHA-256.
    \item \textbf{EC2 $\to$ RDS PostgreSQL:} Worker ghi bản ghi kiểm toán vào bảng \texttt{audit\_logs}.
    \item \textbf{EC2 $\to$ Amazon S3:} Đẩy file kết quả lên S3 \texttt{output/} và sinh S3 Pre-signed URL có thời hạn 15 phút.
    \item \textbf{EC2 $\to$ ALB $\to$ Client:} Trả về HTTP 200 OK kèm JSON chứa mã SHA-256, thời gian xử lý và link tải.
    \item \textbf{Client $\to$ Amazon S3:} Người dùng tải file trực tiếp từ S3, không chiếm dụng băng thông của EC2 hay ALB.
\end{enumerate}

\section{Đặc tả chi tiết kỹ thuật trong giải pháp}

\subsection{Mô hình Lưu trữ đa tầng và Lý do bắt buộc dùng EFS thay vì EBS}
Hệ thống kết hợp cả 03 công nghệ lưu trữ cốt lõi của AWS: \textbf{EBS (gp3)} cho hệ điều hành và thư mục tạm \texttt{/tmp}, \textbf{Amazon S3} cho lưu trữ đối tượng bền vững kèm S3 Lifecycle 24h, và \textbf{Amazon EFS} làm bộ đệm dùng chung (\textit{Shared Cache}) cho toàn cụm.

Trong kiến trúc Auto Scaling đa vùng (Multi-AZ) đứng sau ALB, giải pháp lưu cache trên EBS của từng máy EC2 bộc lộ 03 nhược điểm nghiêm trọng:
\begin{enumerate}
    \item \textbf{Cache Miss \& Cold-start do ALB Round-Robin:} Khi Worker A đã cache file về EBS nhưng request tiếp theo được ALB điều hướng sang Worker B, Worker B lại phải tải lại từ S3 (trễ 1--2 giây).
    \item \textbf{Bất đồng bộ dữ liệu và lỗi Flapping:} Khi tài liệu gốc bị cập nhật hoặc xóa tại Master Library, lệnh xóa chỉ tới 1 worker; các worker khác vẫn giữ file cũ trên EBS, gây ra lỗi hiển thị chập chờn (Flapping).
    \item \textbf{Nhân bản dung lượng:} Cụm 4 máy sẽ nhân 4 lần dung lượng lưu trữ trên EBS.
\end{enumerate}
Do đó, \textbf{Amazon EFS (POSIX ReadWriteMany qua NFSv4)} là lựa chọn duy nhất đảm bảo \textit{Single Source of Truth}, cho phép toàn bộ cụm EC2 truy cập chung bộ đệm với độ trễ $<5$ ms và đảm bảo tính nhất quán dữ liệu tuyệt đối (Strong Consistency).

\subsection{Thiết kế Cơ sở dữ liệu kiểm toán (RDS PostgreSQL Schema)}
Bảng \texttt{audit\_logs} được thiết kế tối ưu hóa tốc độ tra cứu kiểm định:
\begin{lstlisting}[language=SQL]
CREATE TABLE audit_logs (
    id SERIAL PRIMARY KEY,
    request_id VARCHAR(64) NOT NULL,
    client_id VARCHAR(64) DEFAULT 'default',
    doc_name VARCHAR(255) NOT NULL,
    pages_count INTEGER DEFAULT 0,
    sha256 VARCHAR(64) NOT NULL,
    worker_host VARCHAR(100) NOT NULL,
    output_key VARCHAR(255),
    processing_time_ms FLOAT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    student_id VARCHAR(64),
    full_name VARCHAR(128),
    doc_code VARCHAR(128),
    metadata_json TEXT
);
CREATE INDEX idx_audit_logs_sha256 ON audit_logs(sha256);
CREATE INDEX idx_audit_logs_client ON audit_logs(client_id);
\end{lstlisting}

\subsection{Cơ chế phân quyền IAM và Tường lửa Security Groups}
\begin{itemize}
    \item \textbf{ALB Security Group:} Chỉ mở Inbound cổng 80 từ toàn mạng (\texttt{0.0.0.0/0}).
    \item \textbf{EC2 App Security Group:} Tuyệt đối không mở cổng ra Internet; chỉ chấp nhận Inbound cổng 8000 từ ALB Security Group.
    \item \textbf{RDS Security Group:} Chỉ chấp nhận Inbound cổng 5432 từ EC2 App Security Group.
    \item \textbf{EFS Security Group:} Chỉ chấp nhận Inbound cổng 2049 (NFS) từ EC2 App Security Group.
    \item \textbf{IAM Role (\texttt{EC2-PDFWatermark-Role}):} Cấp quyền đọc/ghi đúng S3 bucket \texttt{uet-pdf-watermark} và ghi log CloudWatch. Không lưu trữ bất kỳ access key nào trong code.
\end{itemize}

\subsection{Đặc tả Nền tảng Web Trực tiếp \& RESTful API}
\begin{itemize}
    \item \textbf{Kho tài liệu gốc EFS/S3 (\texttt{POST /api/v1/documents/upload}):} Người dùng chỉ cần tải lên tệp PDF một lần duy nhất lên EFS cache và S3; các lần đóng dấu sau tái sử dụng trực tiếp qua \texttt{doc\_code} với thời gian nạp $<5$ms mà không cần upload lại file.
    \item \textbf{Phòng đóng dấu trực quan \& Canvas Preview (\texttt{POST /api/v1/documents/\{doc\_code\}/watermark}):} Tùy biến dấu chìm, tiêu đề, con dấu và trực quan hóa trang in trực tiếp trên HTML5 Canvas.
    \item \textbf{Đóng dấu hàng loạt đa luồng \& Xuất ZIP (\texttt{POST /api/v1/documents/\{doc\_code\}/batch}):} Xử lý song song đa luồng $N$ bản PDF, lập bảng kê \texttt{kiem\_dinh\_manifest.csv} và nén toàn bộ thành tệp ZIP tải về một chạm.
    \item \textbf{Kiểm định tài liệu Web Crypto API (\texttt{GET /api/v1/audit-logs/verify/\{sha256\}}):} Tính mã băm SHA-256 ngay trên trình duyệt trong 20ms và đối soát cơ sở dữ liệu RDS PostgreSQL.
\end{itemize}
